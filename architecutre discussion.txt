use spotledger_types::{document::Document, error::SpotError};

pub fn require_str<'a>(
    params: &'a HashMap<String, Value>,
    key: &str,
) -> Result<&'a str, SpotError> {
    params
        .get(key)
        .and_then(Value::as_str)
        .ok_or_else(|| SpotError::Validation(format!("'{key}' is required")))
}

pub fn chrono_now() -> String {
    chrono::Utc::now().format("%Y-%m-%d").to_string()
}

/// Resolve Link field titles for _link_titles in getdoc response.
pub async fn build_link_titles(db: &DbAdapter, doctype: &str, doc: &Document) -> Value {
    let filter = serde_json::json!({
        "parent": doctype,
        "parenttype": "DocType",
        "fieldtype": "Link"
    });
    let link_fields = match get_list(db, "DocField", Some(&["fieldname", "options"]), Some(&filter), 200, 0).await {
        Ok(r) => r,
        Err(_) => return serde_json::json!({}),
    };

    let mut titles = serde_json::Map::new();

    for field_row in &link_fields {
        let Some(fieldname) = field_row.get("fieldname").and_then(Value::as_str) else { continue };
        let Some(linked_dt) = field_row.get("options").and_then(Value::as_str) else { continue };
        let Some(val) = doc.get_str(fieldname).filter(|s| !s.is_empty()) else { continue };

        let Ok(dt_doc) = get_doc(db, "DocType", linked_dt).await else { continue };

        let show = dt_doc.get_value("show_title_field_in_link")
            .map(|v| match v {
                Value::Bool(b) => b,
                Value::Number(n) => n.as_i64().unwrap_or(0) != 0,
                _ => false,
            })
            .unwrap_or(false);

        if !show { continue; }

        let Some(title_field) = dt_doc.get_str("title_field").filter(|s| !s.is_empty()) else { continue };

        if let Ok(Some(title)) = get_value(db, linked_dt, val, title_field).await {
            let title_str = match &title {
                Value::String(s) => s.clone(),
                other => other.to_string(),
            };
            titles.insert(format!("{linked_dt}::{val}"), Value::String(title_str));
        }
    }

    Value::Object(titles)
}

/// Fetch User docs and return frappe user_info shape.
pub async fn build_user_info(db: &DbAdapter, users: &[String]) -> Value {
    if users.is_empty() { return serde_json::json!({}); }

    let mut map = serde_json::Map::new();
    for username in users {
        if username.is_empty() { continue; }
        let Ok(user_doc) = get_doc(db, "User", username).await else { continue };
        map.insert(username.clone(), serde_json::json!({
            "fullname":  user_doc.get_str("full_name").unwrap_or(username),
            "image":     user_doc.get_str("user_image").unwrap_or(""),
            "name":      username,
            "email":     user_doc.get_str("email").unwrap_or(""),
            "time_zone": user_doc.get_str("time_zone").unwrap_or(""),
        }));
    }
    Value::Object(map)
}

use serde_json::{json, Value};
use spotledger_db::{DbAdapter, document::{get_doc, get_list}};
use spotledger_types::error::SpotError;
use crate::state::SiteState;
use super::helpers::require_str;
use std::collections::HashMap;
use std::sync::Arc;

pub async fn handle_getdoctype(
    site: Arc<SiteState>,
    params: HashMap<String, Value>,
) -> Result<Value, SpotError> {
    let doctype = require_str(&params, "doctype")?;
    let mut dt_obj = get_doc(&site.db, "DocType", doctype)
        .await
        .map_err(SpotError::from)?
        .as_dict();

    append_doctype_children(&site.db, doctype, &mut dt_obj).await;

    // Collect child doctype names
    let child_names = extract_table_child_names(&dt_obj);

    let mut docs = vec![dt_obj];
    for child_name in child_names {
        if let Ok(child_raw) = get_doc(&site.db, "DocType", &child_name).await {
            let mut child_obj = child_raw.as_dict();
            append_doctype_children(&site.db, &child_name, &mut child_obj).await;
            docs.push(child_obj);
        }
    }

    Ok(json!({ "docs": docs, "lang_modified": null }))
}

async fn append_doctype_children(db: &DbAdapter, dt_name: &str, doc_obj: &mut Value) {
    let filter = json!({ "parent": dt_name, "parenttype": "DocType" });

    let fields  = fetch_child_list(db, "DocField", &filter, 500).await;
    let perms   = fetch_child_list(db, "DocPerm",  &filter, 100).await;
    let states  = fetch_child_list(db, "DocState", &filter, 50).await;

    let Value::Object(ref mut map) = doc_obj else { return };

    map.insert("fields".into(),  Value::Array(fields));
    map.insert("states".into(),  Value::Array(states));
    map.insert("permissions".into(), if perms.is_empty() {
        default_permissions()
    } else {
        Value::Array(perms)
    });
    map.entry("links".to_string()).or_insert(Value::Array(vec![]));
    map.entry("actions".to_string()).or_insert(Value::Array(vec![]));
}

async fn fetch_child_list(db: &DbAdapter, doctype: &str, filter: &Value, limit: usize) -> Vec<Value> {
    get_list(db, doctype, None, Some(filter), limit, 0)
        .await
        .unwrap_or_default()
        .into_iter()
        .filter_map(|r| serde_json::to_value(r).ok())
        .collect()
}

fn extract_table_child_names(dt_obj: &Value) -> Vec<String> {
    let Value::Array(fields) = dt_obj.get("fields").unwrap_or(&Value::Array(vec![])) else {
        return vec![];
    };
    fields.iter()
        .filter(|f| matches!(
            f.get("fieldtype").and_then(Value::as_str),
            Some("Table") | Some("Table MultiSelect")
        ))
        .filter_map(|f| f.get("options").and_then(Value::as_str))
        .filter(|s| !s.is_empty())
        .map(str::to_string)
        .collect()
}

use spotledger_types::error::SpotError;
use spotledger_db::document::*;
use crate::state::SiteState;
use super::helpers::require_str;
use std::collections::HashMap;

pub async fn handle_savedocs(
    site: Arc<SiteState>,
    params: HashMap<String, Value>,
) -> Result<Value, SpotError> {
    let doc_val = parse_doc_param(&params)?;
    let action  = params.get("action").and_then(Value::as_str).unwrap_or("Save");
    let user    = params.get("__current_user").and_then(Value::as_str).unwrap_or("Administrator");
    let doctype = required_str_field(&doc_val, "doctype")?;

    let is_new      = detect_is_new(&doc_val);
    let local_name  = detect_local_name(&doc_val, is_new);
    let saved_name  = resolve_name(&site.db, &doc_val, &doctype, is_new).await?;

    let (parent_val, child_rows) = split_parent_and_children(&doc_val, &saved_name, user);

    apply_action(&site, action, &doctype, &saved_name, &parent_val, is_new).await?;
    save_child_rows(&site.db, &child_rows, &saved_name, &doctype).await;

    let saved_doc   = get_doc(&site.db, &doctype, &saved_name).await
        .map_err(|e| SpotError::Db(e.to_string()))?;
    let perms       = get_doc_permissions(&site.db, user, &doctype).await
        .map(|p| p.to_json())
        .unwrap_or_else(|_| json!({"read": 1, "write": 1, "create": 1}));

    let mut doc_dict = saved_doc.as_dict();
    if let (Some(ln), Value::Object(ref mut map)) = (local_name, &mut doc_dict) {
        map.insert("localname".into(), Value::String(ln));
    }

    Ok(json!({
        "docs": [doc_dict],
        "docinfo": minimal_docinfo(&doctype, &saved_name, perms),
    }))
}

    let doctype = require_str(&params, "doctype")?.to_string();
    let name    = require_str(&params, "name")?.to_string();

    cancel_doc(&site.db, &doctype, &name).await
        .map_err(|e| SpotError::Db(e.to_string()))?;
    site.doc_cache.remove(&(doctype, name.clone())).await;

    Ok(json!({ "docname": name }))
}

fn detect_is_new(doc: &Value) -> bool {
    doc.get("__newname").and_then(Value::as_str).filter(|s| !s.is_empty()).is_none()
    && (
        doc.get("name").and_then(Value::as_str).map(|n| n.starts_with("new-")).unwrap_or(false)
        || doc.get("name").and_then(Value::as_str).map(|n| n.is_empty()).unwrap_or(true)
        || doc.get("__islocal").and_then(Value::as_i64).unwrap_or(0) == 1
    )
}

async fn resolve_name(
    db: &spotledger_db::DbAdapter,
    doc: &Value,
    doctype: &str,
    is_new: bool,
) -> Result<String, SpotError> {
    // Explicit user-supplied name
    if let Some(n) = doc.get("__newname").and_then(Value::as_str).filter(|s| !s.is_empty()) {
        return Ok(n.to_string());
    }

    if !is_new {
        return doc.get("name")
            .and_then(Value::as_str)
            .filter(|s| !s.is_empty())
            .map(str::to_string)
            .ok_or_else(|| SpotError::Validation("name missing for existing doc".into()));
    }

    // Look up autoname strategy from DocType
    let autoname = get_doc(db, "DocType", doctype).await
        .ok()
        .and_then(|dt| dt.get_str("autoname").map(str::to_string));

    match autoname.as_deref() {
        Some(an) if an.starts_with("field:") => {
            let fieldname = &an["field:".len()..];
            doc.get(fieldname)
                .and_then(Value::as_str)
                .filter(|s| !s.is_empty())
                .map(str::to_string)
                .ok_or_else(|| SpotError::Validation(
                    format!("autoname field '{fieldname}' is empty or missing")
                ))
        }
        _ => Ok(generate_random_name(doctype)),
    }
}

fn generate_random_name(doctype: &str) -> String {
    let ts = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .unwrap_or_default()
        .subsec_nanos();
    format!("{}-{ts:08x}", doctype.to_lowercase().replace(' ', "-"))
}

fn split_parent_and_children(
    doc: &Value,
    saved_name: &str,
    current_user: &str,
) -> (Value, Vec<Value>) {
    const SKIP: &[&str] = &[
        "__islocal", "__unsaved", "__newname",
        "__run_link_triggers", "__last_sync_on",
        "__dirty", "__checked", "__doctype_fields",
    ];

    let Value::Object(map) = doc else { return (doc.clone(), vec![]) };

    let mut parent = serde_json::Map::new();
    let mut children = vec![];

    for (k, v) in map {
        if SKIP.contains(&k.as_str()) { continue; }
        if let Value::Array(arr) = v {
            children.extend(arr.iter().filter_map(|r| r.as_object().map(|_| r.clone())));
            continue;
        }
        parent.insert(k.clone(), v.clone());
    }

    parent.insert("name".into(),        saved_name.into());
    parent.entry("owner".into())        .or_insert_with(|| current_user.into());
    parent.entry("modified_by".into())  .or_insert_with(|| current_user.into());
    parent.entry("docstatus".into())    .or_insert(0.into());

    (Value::Object(parent), children)
}

async fn apply_action(
    site: &Arc<SiteState>,
    action: &str,
    doctype: &str,
    name: &str,
    fields: &Value,
    is_new: bool,
) -> Result<(), SpotError> {
    match action {
        "Save" | "Update" => {
            if is_new {
                insert_doc(&site.db, doctype, fields).await
            } else {
                site.doc_cache.remove(&(doctype.to_string(), name.to_string())).await;
                upsert_doc(&site.db, doctype, name, fields).await
            }
            .map_err(|e| SpotError::Db(e.to_string()))
        }
        "Submit" => {
            site.doc_cache.remove(&(doctype.to_string(), name.to_string())).await;
            submit_doc(&site.db, doctype, name).await
                .map_err(|e| SpotError::Db(e.to_string()))
        }
        "Cancel" => {
            site.doc_cache.remove(&(doctype.to_string(), name.to_string())).await;
            cancel_doc(&site.db, doctype, name).await
                .map_err(|e| SpotError::Db(e.to_string()))
        }
        _ => Ok(()),
    }
}

async fn save_child_rows(
    db: &spotledger_db::DbAdapter,
    children: &[Value],
    parent_name: &str,
    parent_doctype: &str,
) {
    for child in children {
        let Some(obj) = child.as_object() else { continue };
        let child_doctype = obj.get("doctype").and_then(Value::as_str).unwrap_or("");
        if child_doctype.is_empty() { continue; }

        let child_name = obj.get("name").and_then(Value::as_str).unwrap_or("").to_string();
        let is_new = child_name.is_empty()
            || child_name.starts_with("new-")
            || obj.get("__islocal").and_then(Value::as_i64).unwrap_or(0) == 1;

        let mut row = child.clone();
        if let Some(m) = row.as_object_mut() {
            m.insert("parent".into(),     parent_name.into());
            m.insert("parenttype".into(), parent_doctype.into());
            m.remove("__islocal");
            m.remove("__unsaved");
            if is_new {
                m.insert("name".into(), generate_random_name(child_doctype).into());
            }
        }

        if is_new {
            let _ = insert_doc(db, child_doctype, &row).await;
        } else {
            let _ = upsert_doc(db, child_doctype, &child_name, &row).await;
        }
    }
}

async fn sync_one_doctype(db: &DbAdapter, meta: &DocTypeMeta) -> Result<()> {
    // 1. Upsert the DocType record itself
    upsert_doc(db, "DocType", &meta.name, &serde_json::to_value(meta)?).await?;

    // 2. Delete old fields, insert fresh ones
    delete_children(db, "DocField", &meta.name).await?;
    for (idx, field) in meta.fields.iter().enumerate() {
        let mut field_val = serde_json::to_value(field)?;
        field_val["parent"]      = meta.name.clone().into();
        field_val["parenttype"]  = "DocType".into();
        field_val["parentfield"] = "fields".into();
        field_val["idx"]         = idx.into();
        insert_doc(db, "DocField", &field_val).await?;
    }

    // 3. DEFINE TABLE / DEFINE FIELD in SurrealDB
    apply_schema(db, meta).await?;

    tracing::info!("Synced DocType: {}", meta.name);
    Ok(())
}

Smart Sync — Only Write What Changed
Don't blindly re-sync everything on every startup. Use a hash:
rustasync fn sync_one_doctype(db: &DbAdapter, meta: &DocTypeMeta) -> Result<()> {
    let json    = serde_json::to_string(meta)?;
    let hash    = format!("{:x}", md5::compute(&json));

    // Check if already synced with same hash
    let stored_hash = get_value(db, "DocType", &meta.name, "_sync_hash")
        .await
        .ok()
        .flatten()
        .and_then(|v| v.as_str().map(str::to_string));

    if stored_hash.as_deref() == Some(&hash) {
        tracing::debug!("DocType {} unchanged, skipping sync", meta.name);
        return Ok(());
    }

    // Hash changed or new — do the full sync
    do_sync(db, meta).await?;

    // Store new hash
    set_field(db, "DocType", &meta.name, "_sync_hash", hash.into()).await?;
    Ok(())
}
Startup goes from "sync 500 doctypes" to "check 500 hashes, sync 2 that changed". Essentially instant.

impl<'a> DocGraph<'a> {
    /// Follow a Link field to its target document
    pub async fn follow<T: DocType + DeserializeOwned>(
        &self,
        link_field: &str,
    ) -> Result<T> {
        let name = self.doc.get_str(link_field)
            .ok_or_else(|| anyhow!("Field {link_field} is empty"))?;
        frappe::get_doc_typed::<T>(name).await
    }

    /// Get all documents that link TO this document
    pub async fn linked_from<T: DocType + DeserializeOwned>(
        &self,
        link_field: &str,
    ) -> Result<Vec<T>> {
        frappe::get_list(T::doctype_name())
            .filter(link_field, "=", &self.doc.name)
            .run_typed::<T>()
            .await
    }

    /// Create a mapped document (like frappe.model.mapper)
    pub async fn make_mapped<S, T>(&self, mapping: &DocMapping<S, T>) -> Result<T>
    where
        S: DocType,
        T: DocType + Default,
    {
        mapping.apply(self.doc).await
    }
}

Are Rhai Scripts Compiled?
Yes — and this is where it gets interesting:
rust// Rhai has THREE execution modes:

// Mode 1 — Parse + run every time (slow, for dev)
engine.eval::<()>(script_source)?;

// Mode 2 — Compile to AST once, run many times (fast)
let ast: AST = engine.compile(script_source)?;
engine.run_ast(&ast)?;   // AST reused, no reparse

// Mode 3 — Compile to AST + optimize (fastest)
let ast = engine.compile_with_scope(&scope, script_source)?;
// Rhai's optimizer does constant folding, dead code elimination
The AST is the compiled form. Not machine code like Rust, but close to bytecode — no parsing on execution, expressions pre-evaluated where possible.

The Clean Architecture
Store scripts in DB, compile to AST on load, cache the AST:
rust// frappe_rhai/engine.rs

pub struct ScriptEngine {
    engine: Engine,
    // AST cache — compiled once, reused forever
    cache:  DashMap<String, Arc<AST>>,
}

impl ScriptEngine {
    pub fn new() -> Self {
        let mut engine = Engine::new();
        register_frappe_api(&mut engine);  // expose frappe::get_doc etc
        Self { engine, cache: DashMap::new() }
    }

    /// Load script from DB, compile, cache
    pub async fn load(&self, doctype: &str, hook: &str, db: &DbAdapter) -> Result<()> {
        let key    = format!("{doctype}::{hook}");
        let source = fetch_script(db, doctype, hook).await?;
        let ast    = self.engine.compile(&source)
            .map_err(|e| anyhow!("Script compile error in {key}: {e}"))?;
        self.cache.insert(key, Arc::new(ast));
        Ok(())
    }

    /// Run cached AST — no recompile
    pub async fn run_hook(
        &self,
        doctype: &str,
        hook: &str,
        doc: &mut Document,
    ) -> Result<()> {
        let key = format!("{doctype}::{hook}");
        let Some(ast) = self.cache.get(&key) else {
            return Ok(());  // no script for this hook — fine
        };

        let mut scope = Scope::new();
        scope.push("doc", PyDoc::from(doc));  // doc is accessible in script

        self.engine.run_ast_with_scope(&mut scope, &ast)?;

        // Write any mutations back
        if let Some(mutated) = scope.get_value::<PyDoc>("doc") {
            mutated.apply_to(doc);
        }

        Ok(())
    }

    /// Hot reload — recompile one script without restart
    pub async fn reload(&self, doctype: &str, hook: &str, db: &DbAdapter) -> Result<()> {
        self.cache.remove(&format!("{doctype}::{hook}"));
        self.load(doctype, hook, db).await
    }
}

    pub async fn load_all(&self, db: &DbAdapter) -> Result<()> {
        let scripts = get_list(db, "Server Script",
            Some(&["name", "doctype_event", "hook", "script"]),
            Some(&json!({"enabled": 1})),
            10_000, 0,
        ).await?;

        let mut errors = vec![];

        for row in &scripts {
            let doctype = row.get_str("doctype_event");
            let hook    = row.get_str("hook");
            let source  = row.get_str("script");
            let key     = format!("{doctype}::{hook}");

                Ok(ast) => { self.cache.insert(key, Arc::new(ast)); }
                Err(e)  => errors.push(format!("{key}: {e}")),
            }
        }

        if !errors.is_empty() {
            // Fail at startup — better than silent wrong behavior
            anyhow::bail!("Script compile errors:\n{}", errors.join("\n"));
        }

        tracing::info!("Compiled {} scripts", self.cache.len());
        Ok(())
    }

    // frappe.get_doc(doctype, name) → doc
    engine.register_fn("get_doc", |doctype: &str, name: &str| {
        block_on(frappe::get_doc(doctype, name))
            .map(RhaiDoc::from)
            .map_err(|e| e.to_string().into())
    });

    // frappe.get_value(doctype, name, field) → value
    engine.register_fn("get_value", |doctype: &str, name: &str, field: &str| {
        block_on(frappe::get_value(doctype, name, field))
            .unwrap_or(None)
    });

    // frappe.get_list(doctype) → builder
    engine.register_fn("get_list", |doctype: &str| {
        RhaiListBuilder::new(doctype)
    });

    // frappe.throw(msg) — validation error
    engine.register_fn("throw", |msg: &str| -> Result<(), Box<EvalAltResult>> {
        Err(format!("ValidationError: {msg}").into())
    });

    // frappe.db.set_value
    engine.register_fn("db_set", |doctype: &str, name: &str, field: &str, val: Dynamic| {
        block_on(frappe::set_value(doctype, name, field, dynamic_to_value(val)))
    });

    // doc methods
    engine.register_type::<RhaiDoc>()
        .register_get_set("name",
            |d: &mut RhaiDoc| d.get_str("name"),
            |d: &mut RhaiDoc, v: String| d.set("name", v))
        .register_fn("get",    |d: &mut RhaiDoc, f: &str| d.get(f))
        .register_fn("set",    |d: &mut RhaiDoc, f: &str, v: Dynamic| d.set_dynamic(f, v))
        .register_fn("save",   |d: &mut RhaiDoc| block_on(d.save()))
        .register_fn("submit", |d: &mut RhaiDoc| block_on(d.submit()))
        .register_fn("append", |d: &mut RhaiDoc, table: &str, row: Map| d.append(table, row));
}

Host Side — Core Loads Plugins
rust// frappe_core/plugins/mod.rs

use extism::{Plugin, Manifest, Wasm, host_fn, UserData};

pub struct PluginRegistry {
    plugins: DashMap<String, Arc<Mutex<Plugin>>>,
}

impl PluginRegistry {
    /// Load all .wasm files from plugins directory
    pub async fn load_all(plugins_dir: &Path) -> Result<Self> {
        let registry = Self { plugins: DashMap::new() };

        for entry in fs::read_dir(plugins_dir)? {
            let path = entry?.path();
            if path.extension() != Some("wasm".as_ref()) { continue; }

            let name = path.file_stem()
                .and_then(|s| s.to_str())
                .unwrap_or("unknown")
                .to_string();

            let manifest = Manifest::new([Wasm::file(&path)]);
            let plugin   = Plugin::new(&manifest, [frappe_host_functions()], true)?;

            registry.plugins.insert(name.clone(), Arc::new(Mutex::new(plugin)));
            tracing::info!("Loaded plugin: {name}");
        }

        Ok(registry)
    }

    /// Call a hook on all plugins that handle it
    pub async fn dispatch_hook(
        &self,
        doctype: &str,
        hook: &str,
        doc: &mut Document,
    ) -> Result<()> {
        let fn_name = format!("{}__{}", doctype.replace(' ', "_"), hook);

        for entry in &self.plugins {
            let mut plugin = entry.value().lock().unwrap();

            // Plugin only called if it exports this function
            if !plugin.function_exists(&fn_name) { continue; }

            let input  = serde_json::to_vec(doc)?;
            let output = plugin.call::<&[u8], &[u8]>(&fn_name, &input)?;
            let updated: Document = serde_json::from_slice(output)?;
            *doc = updated;
        }

        Ok(())
    }

    /// Register doctypes from all plugins
    pub async fn register_doctypes(&self, registry: &DocTypeRegistry) -> Result<()> {
        for entry in &self.plugins {
            let mut plugin = entry.value().lock().unwrap();
            if !plugin.function_exists("register_doctypes") { continue; }

            let output   = plugin.call::<(), &[u8]>("register_doctypes", ())?;
            let metas: Vec<DocTypeMeta> = serde_json::from_slice(output)?;

            for meta in metas {
                registry.register_dynamic(meta.into());
                tracing::info!("Registered doctype: {} from plugin {}", meta.name, entry.key());
            }
        }
        Ok(())
    }
}

Migration Runner — Core Side
Core runs migrations for every plugin on startup:
rust// frappe_core/migrations/runner.rs

pub struct MigrationRunner<'a> {
    db:      &'a DbAdapter,
    plugins: &'a PluginRegistry,
}

impl<'a> MigrationRunner<'a> {
    pub async fn run_all(&self) -> Result<()> {
        // Ensure migration tracking table exists
        self.ensure_migration_table().await?;

        for (plugin_name, plugin) in &self.plugins.plugins {
            self.run_plugin_migrations(plugin_name, plugin).await?;
        }

        Ok(())
    }

    async fn run_plugin_migrations(
        &self,
        plugin_name: &str,
        plugin: &Mutex<Plugin>,
    ) -> Result<()> {
        // Ask plugin for its migrations
        let migrations: Vec<Migration> = {
            let mut p = plugin.lock().unwrap();
            if !p.function_exists("migrations") {
                return Ok(());  // plugin has no migrations, skip
            }
            let output = p.call::<(), Json<Vec<Migration>>>("migrations", ())?;
            output.0
        };

        // What version is this plugin currently at?
        let current_version = self.get_current_version(plugin_name).await?;
        let pending: Vec<_> = migrations.iter()
            .filter(|m| m.version > current_version)
            .collect();

        if pending.is_empty() {
            tracing::debug!("Plugin {plugin_name}: no migrations needed (v{current_version})");
            return Ok(());
        }

        tracing::info!(
            "Plugin {plugin_name}: running {} migrations (v{} → v{})",
            pending.len(),
            current_version,
            pending.last().unwrap().version,
        );

        // Run each pending migration in order
        for migration in pending {
            tracing::info!(
                "  [{plugin_name}] v{}: {}",
                migration.version,
                migration.description
            );

            // Execute the SurrealQL
            self.db.execute_raw(migration.up).await
                .map_err(|e| anyhow!(
                    "Migration failed [{plugin_name} v{}]: {e}\nSQL:\n{}",
                    migration.version,
                    migration.up
                ))?;

            // Record success
            self.record_migration(plugin_name, migration).await?;
        }

        Ok(())
    }

    async fn ensure_migration_table(&self) -> Result<()> {
        self.db.execute_raw("
            DEFINE TABLE IF NOT EXISTS _frappe_migrations SCHEMAFULL;
            DEFINE FIELD plugin      ON _frappe_migrations TYPE string;
            DEFINE FIELD version     ON _frappe_migrations TYPE int;
            DEFINE FIELD description ON _frappe_migrations TYPE string;
            DEFINE FIELD applied_at  ON _frappe_migrations TYPE datetime;
            DEFINE INDEX idx_plugin_version
                ON _frappe_migrations FIELDS plugin, version UNIQUE;
        ").await
    }

    async fn get_current_version(&self, plugin_name: &str) -> Result<u32> {
        let rows = self.db.run(
            "SELECT version FROM _frappe_migrations
             WHERE plugin = $plugin
             ORDER BY version DESC
             LIMIT 1",
            vec![("plugin".into(), plugin_name.into())],
        ).await?;

        Ok(rows.first()
            .and_then(|r| r.get("version"))
            .and_then(|v| v.as_u64())
            .map(|v| v as u32)
            .unwrap_or(0))
    }

    async fn record_migration(
        &self,
        plugin_name: &str,
        migration: &Migration,
    ) -> Result<()> {
        self.db.execute(
            "CREATE _frappe_migrations SET
             plugin      = $plugin,
             version     = $version,
             description = $description,
             applied_at  = time::now()",
            vec![
                ("plugin".into(),      plugin_name.into()),
                ("version".into(),     migration.version.into()),
                ("description".into(), migration.description.into()),
            ],
        ).await
    }
}

Startup Sequence — Full Picture
rust// main.rs

#[tokio::main]
async fn main() -> Result<()> {
    let db = Arc::new(DbAdapter::connect(&config.db_url).await?);

    // 1. Load plugins
    let plugins = Arc::new(
        PluginRegistry::load_all(Path::new("./plugins")).await?
    );

    // 2. Run migrations for all plugins
    // Fails loud if any migration errors — don't start broken
    MigrationRunner { db: &db, plugins: &plugins }
        .run_all()
        .await
        .context("Migration failed — fix the error and restart")?;

    // 3. Register doctypes from plugins into registry
    let registry = Arc::new(DocTypeRegistry::init(&db).await?);
    plugins.register_doctypes(&registry).await?;

    // 4. Init frappe context
    frappe_core::init_with(db, registry, plugins).await?;

    // 5. Serve
    tracing::info!("All migrations applied. Starting server.");
    serve().await
}

What Happens When You Update a Plugin
erpnext v1.0 → v2.0: added delivery_date field

1. cargo build --package erpnext  (AI does this)
   → new erpnext.wasm contains migration v2

2. cp erpnext-v2.wasm plugins/erpnext.wasm

3. systemctl restart frappe-rs

4. On startup:
   PluginRegistry loads erpnext.wasm
   MigrationRunner asks: what's current version? → 1
   MigrationRunner asks plugin: what migrations exist? → [v1, v2]
   Runs v2: DEFINE FIELD delivery_date ON tabSalesOrder...
   Records v2 applied
   Server starts

Total downtime: ~2 seconds (restart)
Zero manual SQL. Zero bench migrate. Zero Python.

Rollback
rustimpl MigrationRunner<'_> {
    pub async fn rollback_plugin(&self, plugin_name: &str, to_version: u32) -> Result<()> {
        let current = self.get_current_version(plugin_name).await?;

        // Get migrations to undo in reverse order
        let migrations = self.get_plugin_migrations(plugin_name).await?;
        let to_undo: Vec<_> = migrations.iter()
            .filter(|m| m.version > to_version && m.version <= current)
            .rev()  // reverse order for rollback
            .collect();

        for migration in to_undo {
            match migration.down {
                Some(sql) => {
                    tracing::info!("Rolling back [{plugin_name}] v{}", migration.version);
                    self.db.execute_raw(sql).await?;
                    self.remove_migration_record(plugin_name, migration.version).await?;
                }
                None => return Err(anyhow!(
                    "Migration v{} has no rollback SQL — cannot proceed",
                    migration.version
                )),
            }
        }
        Ok(())
    }
}

The Full Picture

    exports:
      migrations()           → Vec<Migration> with embedded SurrealQL
      register_doctypes()    → Vec<DocTypeMeta>
      Sales_Order__validate()
      Sales_Invoice__on_submit()

  hrms.wasm
    exports:
      migrations()           → its own migrations, namespaced
      register_doctypes()    → Leave, Payroll, Employee
      Employee__validate()
      ...

Core on startup:
  for each .wasm:
    1. run pending migrations   (schema changes)
    2. register doctypes        (UI/API schema)
    3. register hooks           (business logic)

Key Design Decisions
Migration SQL embedded in .wasm    → no external files, self-contained plugin
_frappe_migrations tracks by plugin → plugins don't interfere with each other
Startup fails if migration fails   → never run on broken schema
down SQL optional                  → irreversible migrations are fine, just say so
AI compiles on field change        → recompile cost = irrelevant
The migration system is actually cleaner than Frappe's — each plugin owns its migrations, they're versioned with the code in git, embedded in the binary, and run automatically. No bench migrate, no manual SQL, no drift between code and schema.
